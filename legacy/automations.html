from spindle import project as sp
import pandas as pd

results = sp.files.read("results/latest.csv")
baseline = sp.files.read("baselines/survey.csv")

outliers = results[results.zscore > 2.5]
digest = sp.agent.summarize(outliers, baseline)

sp.log.write("notes.md", digest)
sp.notify("#study-updates", digest)
